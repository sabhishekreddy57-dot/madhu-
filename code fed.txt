/* =====================================================
   SUBMIT FOOD
===================================================== */


function submitFood() {


    const donorName =
        document.getElementById("donorName")
        .value.trim();


    const foodName =
        document.getElementById("foodName")
        .value.trim();


    const phone =
        document.getElementById("foodPhone")
        .value.trim();


    const quantity =
        document.getElementById("foodQuantity")
        .value.trim();


    const location =
        document.getElementById("foodLocation")
        .value.trim();


    const expiry =
        document.getElementById("foodExpiry")
        .value;



    if (
        !donorName ||
        !foodName ||
        !phone ||
        !quantity ||
        !location ||
        !expiry
    ) {

        alert("Please fill all food details.");

        return;
    }



    /* GET EXISTING FOOD */

    let foods =
        JSON.parse(
            localStorage.getItem("foodPosts")
        ) || [];



    /* CREATE FOOD */

    const food = {

        id: Date.now(),

        donorName: donorName,

        foodName: foodName,

        phone: phone,

        quantity: quantity,

        location: location,

        expiry: expiry,

        status: "Pending Admin Verification",

        recipientName: "",

        recipientPhone: "",

        recipientLocation: "",

        claimStatus: "Not Claimed"

    };



    /* ADD FOOD */

    foods.push(food);



    /* SAVE FOOD */

    localStorage.setItem(
        "foodPosts",
        JSON.stringify(foods)
    );



    alert(
        "Food details sent to Admin successfully! 🍱"
    );



    /* CLEAR FORM */

    document.getElementById("donorName").value = "";

    document.getElementById("foodName").value = "";

    document.getElementById("foodPhone").value = "";

    document.getElementById("foodQuantity").value = "";

    document.getElementById("foodLocation").value = "";

    document.getElementById("foodExpiry").value = "";



    displayDonorFood();

}

                    <span class="status

/* =====================================================

===================================================== */


function loadAdminFood() {


    const list =
        document.getElementById("adminFoodList");


    const foods =
        JSON.parse(
            localStorage.getItem("foodPosts")
        ) || [];



    if (foods.length === 0) {

        list.innerHTML =
            "<p>No food has been submitted by donors.</p>";

        return;

    }



    list.innerHTML = "";



    foods.forEach((food, index) => {


        let buttons = "";



        /* PENDING FOOD */

        if (
            food.status ===
            "Pending Admin Verification"
        ) {

            buttons = `

                <button
                    onclick="verifyFood(${index})">

                    ✅ Verify Food

                </button>


                <button
                    class="reject"
                    onclick="rejectFood(${index})">

                    ❌ Reject Food

                </button>

            `;

        }



        /* FOOD DETAILS */

        list.innerHTML += `

            <div class="food">

                <h3>
                    🍱 ${food.foodName}
                </h3>

                <p>
                    <b>Donor:</b>
                    ${food.donorName}
                </p>

                <p>
                    <b>Phone:</b>
                    ${food.phone}
                </p>

                <p>
                    <b>Quantity:</b>
                    ${food.quantity}
                </p>

                <p>
                    <b>Location:</b>
                    ${food.location}
                </p>

                <p>
                    <b>Expiry:</b>
                    ${food.expiry}
                </p>

                <p>
                    <b>Status:</b>

                    <span class="status
                    ${getStatusClass(food.status)}">

                        ${food.status}

                    </span>

                </p>


                ${
                    food.claimStatus === "Claimed"

                    ?

                    `

                    <div class="info">

                        <b>📦 Food Claimed by Recipient</b>

                        <p>
                            <b>Name:</b>
                            ${food.recipientName}
                        </p>

                        <p>
                            <b>Phone:</b>
                            ${food.recipientPhone}
                        </p>

                        <p>
                            <b>Location:</b>
                            ${food.recipientLocation}
                        </p>

                    </div>

                    `

                    :

                    ""

                }


                ${buttons}

            </div>

        `;

    });

}



/* =====================================================
   VERIFY FOOD
===================================================== */


function verifyFood(index) {


    let foods =
        JSON.parse(
            localStorage.getItem("foodPosts")
        ) || [];



    foods[index].status =
        "Available";



    foods[index].claimStatus =
        "Not Claimed";



    localStorage.setItem(
        "foodPosts",
        JSON.stringify(foods)
    );



    alert(
        "Food Verified Successfully! ✅"
    );


    loadAdminFood();

}

/* =====================================================

===================================================== */


function loadAvailableFood() {


    const list =
        document.getElementById("availableFoodList");


    const foods =
        JSON.parse(
            localStorage.getItem("foodPosts")
        ) || [];



    const availableFoods =
        foods.filter(food =>

            food.status === "Available" &&
            food.claimStatus !== "Claimed"

        );



    if (availableFoods.length === 0) {

        list.innerHTML =
            "<p>No verified food is currently available.</p>";

        return;

    }



    list.innerHTML = "";



    foods.forEach((food, index) => {


        if (
            food.status === "Available" &&
            food.claimStatus !== "Claimed"
        ) {


            list.innerHTML += `

                <div class="food">

                    <h3>
                        🍱 ${food.foodName}
                    </h3>

                    <p>
                        Quantity:
                        ${food.quantity}
                    </p>

                    <p>
                        Pickup Location:
                        ${food.location}
                    </p>

                    <p>
                        Expiry:
                        ${food.expiry}
                    </p>

                    <p>
                        Status:

                        <span class="available">

                            ✅ Verified & Available

                        </span>

                    </p>


                    <button
                        class="claim"
                        onclick="claimFood(${index})">

                        📦 Claim Food

                    </button>

                </div>

            `;

        }

    });

}



/* =====================================================
   CLAIM FOOD
===================================================== */


function claimFood(index) {


    const recipientName =
        document.getElementById("recipientName")
            .value.trim();


    const recipientPhone =
        document.getElementById("recipientPhone")
            .value.trim();


    const recipientLocation =
        document.getElementById("recipientLocation")
            .value.trim();



    /* CHECK RECIPIENT DETAILS */

    if (
        !recipientName ||
        !recipientPhone ||
        !recipientLocation
    ) {

        alert(
            "Please enter your name, phone and location before claiming."
        );

        return;

    }



    /* GET FOOD */

    let foods =
        JSON.parse(
            localStorage.getItem("foodPosts")
        ) || [];



    /* CHECK AVAILABILITY */

    if (
        !foods[index] ||
        foods[index].status !== "Available" ||
        foods[index].claimStatus === "Claimed"
    ) {

        alert(
            "Sorry! This food is no longer available."
        );

        loadAvailableFood();

        return;

    }



    /* SAVE RECIPIENT DETAILS */

    foods[index].recipientName =
        recipientName;


    foods[index].recipientPhone =
        recipientPhone;


    foods[index].recipientLocation =
        recipientLocation;


    foods[index].claimStatus =
        "Claimed";


    foods[index].status =
        "Claimed";



    /* SAVE */

    localStorage.setItem(
        "foodPosts",
        JSON.stringify(foods)
    );



    alert(
        "Food claimed successfully! 📦"
    );


    loadAvailableFood();

    loadRecipientClaims();

}

/* =====================================================
   STATUS CLASS
===================================================== */


function getStatusClass(status) {


    if (status === "Available") {

        return "available";

    }


    if (status === "Rejected") {

        return "rejected";

    }


    if (status === "Claimed") {

        return "claimed";

    }


    return "pending";

}
